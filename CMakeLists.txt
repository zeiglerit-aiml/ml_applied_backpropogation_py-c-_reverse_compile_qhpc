cmake_minimum_required(VERSION 3.20)
project(hybrid_ml_execution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HYBRID_ML_BUILD_PYTHON_BINDINGS "Build pybind11 bindings" OFF)
option(HYBRID_ML_ENABLE_OPENMP "Enable OpenMP where available" ON)

add_library(hybrid_ml_core STATIC cpp/src/ops.cpp)
target_include_directories(hybrid_ml_core PUBLIC cpp/include)
if(HYBRID_ML_ENABLE_OPENMP)
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(hybrid_ml_core PUBLIC OpenMP::OpenMP_CXX)
  endif()
endif()
if(HYBRID_ML_BUILD_PYTHON_BINDINGS)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native cpp/bindings/module.cpp)
  target_link_libraries(_native PRIVATE hybrid_ml_core)
endif()
